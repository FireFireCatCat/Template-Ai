   Continued \relativeframelocation{static}{#2}{static}{#1}% 
 }

   Continued from \relativeframelocation{static}{#1}{static}{#2}% 
 }% 
 \ffdefaultpostcontinued
}

elsewhere on the page.

\end{staticcontents*}
