\documentclass[english]{smfart}

%% Unlike amsart, smfart does not auto-provide theorem environments --
%% declare what you need explicitly.
\newtheorem{theorem}{Theorem}

\title{Your Title Here}
\author{Ada Lovelace}
\address{Institute of Analytical Engines, London, United Kingdom}
\email{ada@example.edu}

%% \subjclass takes ONE mandatory brace group only (plain \def, no xparse
%% optional-argument support) -- amsart's \subjclass[2020]{...} year-bracket
%% syntax does NOT work here: the stray "[2020]" leaks into the preamble as
%% ordinary text and breaks with "Missing \begin{document}".
\subjclass{Primary 11A41; Secondary 11N05}
\keywords{first keyword, second keyword, third keyword}

\begin{document}

\begin{abstract}
Replace this with your abstract. A good abstract states the problem, what
you did, and what you found --- in that order, and without citations.
\end{abstract}

\maketitle

\section{Introduction}
Start here. Cite things like this~\cite{knuth1984texbook}.

\section{Main results}
\label{sec:results}

\begin{theorem}
\label{thm:main}
State your main result here.
\end{theorem}

\begin{proof}
Prove it here.
\end{proof}

\section{Conclusion}
Say what it means.

\bibliographystyle{smfplain}
\bibliography{references}

\end{document}
