% Multilingual Wall Calendar — a full-page (A4 "giant") 12-month wall
% calendar with a cover page, dated events, and a TikZ illustration per
% month. Built on LianTze Lim's cdcalendar class.
\documentclass[17pt,british,giant]{cdcalendar}

\usepackage[T1]{fontenc}
\usepackage[rm]{roboto}
\usepackage[notextcomp,notext]{stix2}

\usepackage{tikz}
\usetikzlibrary{shapes.geometric}
\usepackage{fontawesome}

\tikzset{blue icon/.style={text=SkyBlue,font=\Large}}
\tikzset{pink icon/.style={text=Pink,font=\large}}
\tikzset{holiday/.style={rectangle,fill=orange!70}}
\tikzset{smfillbg/.style={opacity=0.4,inner sep=1.5ex}}

\geometry{a4paper}

% A small, self-contained TikZ illustration used on a few months below, so
% this starter has zero external image dependencies (upstream's own example
% pulls in raster art from a third-party package). Draws a simple sun-over-
% hills motif; recolour or replace freely.
\newcommand{\monthIllustration}{%
  \begin{tikzpicture}
    \fill[SkyBlue!25] (0,0) rectangle (10,4);
    \fill[yellow!80!orange] (8,3.2) circle (0.8);
    \fill[green!45!black] (-0.5,0) .. controls (2,2.2) and (4,0.3) .. (6,1.6)
      .. controls (7.5,2.4) and (9,0.6) .. (10.5,1.4) -- (10.5,-0.2)
      -- (-0.5,-0.2) -- cycle;
    \fill[green!30!black] (-0.5,-0.2) .. controls (1.5,1.2) and (3,-0.4) .. (5,0.6)
      .. controls (7,1.4) and (8.5,-0.2) .. (10.5,0.5) -- (10.5,-0.2) -- cycle;
  \end{tikzpicture}%
}
