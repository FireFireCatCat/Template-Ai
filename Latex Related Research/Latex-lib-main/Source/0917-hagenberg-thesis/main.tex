%% UPSTREAM BUG WORKAROUND: hgb.sty calls the LaTeX2e-kernel primitive
%% \DocumentMetadata from inside \AtBeginDocument (hgb.sty's own comment
%% even says "This may only be called in the document preamble!" while
%% doing exactly that -- \AtBeginDocument code runs AFTER \begin{document},
%% not in the preamble). \DocumentMetadata is only valid before
%% \documentclass in the current LaTeX2e kernel, so calling it this late
%% throws "should be used only before \documentclass" plus knock-on
%% "already defined" errors from re-entering PDF-management setup. This
%% document does not need PDF metadata/PDF-A support, so the safest fix is
%% to no-op \DocumentMetadata for the one call the class makes internally,
%% rather than patch the vendored class file itself.
\makeatletter
\let\DocumentMetadata\@gobble
\makeatother
