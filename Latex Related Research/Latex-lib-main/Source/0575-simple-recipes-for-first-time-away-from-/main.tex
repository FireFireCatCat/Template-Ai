\quickrecipe{One-pan tomato pasta}{Serves 1}{%
  \textbf{You need:} 90 g pasta, 1 cup tomatoes, 1 garlic clove, olive oil, salt, and basil.\\[2pt]
  \textbf{Do this:} Cook the pasta. Warm garlic and tomatoes in oil, toss with the drained pasta, and top with basil.}
\quickrecipe{Breakfast oats}{Serves 1}{%
  \textbf{You need:} 50 g oats, 200 ml milk, 1 banana, cinnamon, and a spoon of nut butter.\\[2pt]
  \textbf{Do this:} Simmer oats and milk for five minutes. Slice over the banana, add cinnamon, and finish with nut butter.}
\quickrecipe{Crispy bean quesadilla}{Serves 1}{%
  \textbf{You need:} 1 tortilla, 100 g beans, grated cheese, sweetcorn, salsa, and lime.\\[2pt]
  \textbf{Do this:} Mash beans with lime. Fill half the tortilla, fold, and fry on both sides until crisp. Cut into wedges.}
