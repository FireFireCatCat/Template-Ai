% !TEX TS-program = xelatex
%
% Business card sheet: 8 cards (2 columns x 4 rows) imposed on US Letter,
% each card at the standard 3.5in x 2in size.
%
% Based on the single-card design at github.com/opieters/business-card
% (GPL-3.0), reworked into a full imposed sheet: the upstream is one loose
% card specifying Fira Sans / Fira Mono and Font Awesome icon glyphs, none of
% which this image ships (FontAwesome.otf is on TeX Live's disk but not
% registered under fontconfig, so xelatex cannot resolve it by name). Nimbus
% Sans substitutes for Fira Sans (metric-compatible Helvetica clone) and the
% icon column is replaced with plain text labels rather than depending on a
% font this image cannot load.
\documentclass[10pt]{article}

\usepackage[paperwidth=8.5in,paperheight=11in,margin=0.5in,noheadfoot]{geometry}
\usepackage{parskip}
\usepackage{fontspec}
\usepackage{tikz}
\usepackage{xcolor}

\setsansfont{Nimbus Sans}
