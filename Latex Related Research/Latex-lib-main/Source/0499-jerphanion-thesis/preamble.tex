% A left-coloured bar around a theorem-like block — used sparingly, since
% every block having one gets noisy fast.
\renewenvironment{leftbar}[1][black]{%
    \def\FrameCommand{{\color{#1}\vrule width 0.5pt}\hspace{0pt}%
        \fboxsep=\FrameSep\colorbox{lightgray}}%
    \MakeFramed{\hsize\hsize\advance\hsize-\width\FrameRestore}%
}{%
    \endMakeFramed%
}
