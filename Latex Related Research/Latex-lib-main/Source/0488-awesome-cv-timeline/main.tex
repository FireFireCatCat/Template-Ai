% catalogue has.
%
% This starter deliberately OMITS the publications/bibliography feature:
% the upstream examples hardcode `backend=biber` in the class's biblatex
% load, and this pipeline runs classic bibtex, not biber — citations would
% silently never resolve. \addbibresource is never called by the class
% itself (only by the upstream example .tex files), so simply not using it
% avoids the trap entirely rather than patching biblatex's backend.

% \headersmargin and \columnsmargin are used by the upstream multi-pages
% example but never \newlength'd anywhere in resume.cls (only sidemargin,
% leftcolumnwidth and timelinemargin are) — a real upstream omission, not
% a document-level choice. Declared here rather than patching the class,
% since these are genuinely document layout lengths.
\newlength{\headersmargin}
\newlength{\columnsmargin}

% \headersmargin positions the absolutely-placed sidebar/body textblocks
% relative to the TOP of the page (textpos convention) — it must clear
% the actual rendered height of the header + summary above, which varies
% with how many lines they wrap to. The upstream example's -0.2cm assumed
% a single-line, no-summary header; tuned by hand here against this
% starter's actual two-line header + two-line summary.
\setlength{\headersmargin}{3.6cm}
\setlength{\leftcolumnwidth}{5cm}
\setlength{\columnsmargin}{0.5\sidemargin}
\setlength{\timelinemargin}{0.5cm}

% Draws the timeline down the middle of the page, spanning the sections
% that land on the RIGHT side of the layout. \cvsection increments a
% GLOBAL A/B/C/... counter every time it is called, in document order —
% it is not tied to which textblock the section is physically in. Here
% Contact=A, Skills=B (both in the left sidebar), Experience=C,
% Education=D (both in the right body) — so the timeline must start at
% C, the first section on the right, per the class's own \timeline{<letter>}
% convention ("first section on the right side").
\timeline{C}
