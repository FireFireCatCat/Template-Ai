\begin{recipe}{Braised Red Cabbage}
\ingredient{1 medium} red cabbage, shredded
\ingredient{2} apples, peeled and diced
\ingredient{1} onion, sliced
\ingredient{2 tbsp} butter
\ingredient{3 tbsp} red wine vinegar
\ingredient{2 tbsp} sugar
\ingredient{1} bay leaf
\ingredient{to taste} salt and pepper

Melt the butter in a large pot and sweat the onion until soft.
Add the cabbage and apples, then stir in the vinegar and sugar.

Tuck in the bay leaf, cover, and simmer gently for about an hour,
stirring occasionally, until the cabbage is tender and glossy.
Season to taste and remove the bay leaf before serving.

\energy{450}
\sidedish{Roast pork or sausages}
\hint{Tastes even better reheated the next day}
\preparationtime{1 hour 15 minutes}
\modification{Add a handful of raisins for a sweeter finish}
\end{recipe}

\begin{recipe}{Potato Dumplings}
\ingredient{1 kg} floury potatoes
\ingredient{100 g} potato starch
\ingredient{1} egg
\ingredient{to taste} salt
\ingredient{50 g} butter

Boil the potatoes in their skins until tender, then peel and rice
them while still hot.

Work in the starch, egg and salt to form a smooth dough. Shape into
walnut-sized dumplings.

Simmer the dumplings in salted water for about 15 minutes, until
they float. Drain and toss with melted butter before serving.

\preparationtime{50 minutes}
\sidedish{Roast goose or braised beef}
\end{recipe}
