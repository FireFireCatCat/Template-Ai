\section{Why We Counted}

\begin{frame}{The question we started with}
  \begin{itemize}
    \item Otter sightings along the Ouse had \lapis{tripled} in casual reports since 2019
    \item No one had run a systematic transect count in over a decade
    \item \memo[user=PN,color=carandache]{Check whether the 2019 baseline survey is archived anywhere}
  \end{itemize}
  \NB{This deck walks through method, results, and what we would do differently.}
\end{frame}

\section{Method}

\begin{frame}{Six transects, one season}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Where}
      \begin{itemize}
        \item 6 transects, 2km each
        \item Dawn and dusk visits
        \item March through September
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{What we logged}
      \begin{itemize}
        \item Direct sightings
        \item Spraint and footprint sign
        \item Water clarity, temperature
      \end{itemize}
    \end{column}
  \end{columns}
  \vs{3}
  {\hand Six transects felt like a lot in January.}
\end{frame}

\begin{frame}{Confluence stands out}
  \stabilo{Confluence and Mill Race} account for over half of all sign recorded, despite being two of the six transects.
  \begin{itemize}
    \item Both sit downstream of the new reed-bed restoration
    \item Water clarity readings were highest at these two sites all season
    \item Standard ecological theory predicts otter density tracks prey-fish abundance, which we did not measure directly
  \end{itemize}
\end{frame}

\section{What We'd Change}

\begin{frame}{Next season}
  \begin{enumerate}
    \item Add fish-abundance sampling at Confluence and Mill Race
    \item Move Old Sluice to a summer-only visit; six months of near-zero sign was not informative
    \item \handbold{Recruit a second observer for dawn visits} --- fatigue showed in the May and June logs
  \end{enumerate}
\end{frame}
