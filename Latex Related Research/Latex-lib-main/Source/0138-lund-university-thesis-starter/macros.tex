\makeatletter
\def\thickhrulefill{\leavevmode \leaders \hrule height 1ex \hfill \kern \z@}
\def\@makechapterhead#1{%
  \markmargin%
  \vspace*{10\p@}%
  {\parindent \z@ \raggedleft \reset@font
            \sffamily \bfseries %\scshape %\@chapapp{} \thechapter%
        \par\nobreak%
        \interlinepenalty\@M%
    \Huge #1\par\nobreak
    \hrulefill
    \par\nobreak
    \vskip 20\p@
  }}
\def\@makeschapterhead#1{%
  \markmarginKappa%
  \vspace*{10\p@}%
  {\parindent \z@ \raggedleft \reset@font
            \sffamily \bfseries %\scshape %\@chapapp{} \thechapter%
        \par\nobreak
        \interlinepenalty\@M
    \Huge  #1\par\nobreak
    \hrulefill
    \par\nobreak
    \vskip 20\p@
  }}

%------------------------ part page layout
%% from book.cls:
\renewcommand\part{%
   \if@openright
      \cleardoublepage
   \else
      \clearpage
   \fi
   \thispagestyle{plain}%
  \if@twocolumn
    \onecolumn
    \@tempswatrue
  \else
    \@tempswafalse
  \fi
  \null\vfil
  \secdef\@part\@spart}

\def\@part[#1]#2{%
    \ifnum \c@secnumdepth >-2\relax
      \refstepcounter{part}%
      \addcontentsline{toc}{part}{#1}%
    \else
      \addcontentsline{toc}{part}{#1}%
    \fi
    \markboth{}{}%
    {\centering
     \interlinepenalty \@M
     \normalfont
     \Huge \reset@font \hrulefill \\ \raggedleft \vspace{5mm} \sffamily%
            \bfseries {#2} \hspace{15mm}\ \\ \hrulefill \par}%
    \@endpart}
\def\@spart#1{%
    {\centering
     \interlinepenalty \@M
     \normalfont
     \Huge \sffamily \bfseries scshape #1\par}%
    \@endpart}
\def\@endpart{\vfil    %% \newpage -- HWL: no pagebreak with part env
              \if@twoside
                \null
                \thispagestyle{empty}%
                %%\newpage
              \fi
              \if@tempswa
                \twocolumn
              \fi}
