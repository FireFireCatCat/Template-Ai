\ctext{0.1}{0.35}{0.8}{\centering\Large\bfseries Retail Checkout Redesign}
\ctext{0.1}{0.45}{0.8}{\centering\normalsize A/B results from the self-checkout pilot}

\normalpage{Why Redesign}{
  \begin{itemize}
    \item Average checkout time had crept up to 94 seconds
    \item Customers cited the barcode-scan step as the main friction point
    \item A redesigned scan flow was tested against the old one for six weeks
  \end{itemize}
}

\normalpage{Two Flows Compared}{
  \textbf{Old flow}: scan, bag, pay, receipt --- four screens.\par
  \textbf{New flow}: scan-and-bag combined, pay, receipt --- three screens.
}

\normalpage{Checkout Time, Before and After}{
  \begin{tabular}{lcc}
    Store & Old (s) & New (s) \\
    \hline
    Store A & 96 & 71 \\
    Store B & 91 & 68 \\
    Store C & 98 & 74 \\
  \end{tabular}
}

\normalpage{What We Recommend}{
  \begin{enumerate}
    \item Roll the new flow out to all stores
    \item Keep the four-screen flow available as an accessibility option
    \item Re-measure after the holiday volume spike
  \end{enumerate}
}
