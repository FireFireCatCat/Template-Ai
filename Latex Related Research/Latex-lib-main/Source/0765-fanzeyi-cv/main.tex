% Fonts
%
% PATCHED for latextemplates.com: upstream's \newfontfamily calls
% resolved "Raleway"/"Raleway Medium"/etc. by FONTCONFIG FAMILY NAME.
% That name is not registered in the pinned image (confirmed: `fc-list |
% grep -i raleway` returns nothing), even though TeX Live itself ships
% the actual Raleway .otf files (impallari/raleway) — same class of gap
% as the classic `fontawesome` package elsewhere in this catalogue.
% Loading each weight by its exact kpathsea-resolvable filename instead
% of a fontconfig family name sidesteps the lookup entirely.
% Raleway ships inside TeX Live itself (texmf-dist/fonts/opentype/
% impallari/raleway/), resolvable by kpathsea under its exact filename —
% fontspec's [Extension=.otf] tells it to search for that filename via
% kpathsea rather than asking fontconfig to resolve a family name.
\newfontfamily\raleway{Raleway-Regular}[Extension=.otf]
\newfontfamily\ralewaym{Raleway-Medium}[Extension=.otf]
\newfontfamily\ralewaysb{Raleway-SemiBold}[Extension=.otf]
\newfontfamily\ralewayb{Raleway-Bold}[Extension=.otf]
\newfontfamily\ralewayeb{Raleway-ExtraBold}[Extension=.otf]
\newfontfamily\ralewaybb{Raleway-Black}[Extension=.otf]

\setmainfont{Raleway-Regular}[Extension=.otf,
  BoldFont=Raleway-Bold, ItalicFont=Raleway-RegularItalic,
  BoldItalicFont=Raleway-BoldItalic]
