\section{Background}

\begin{frame}{Why Non-IID Data Is Hard}
  \begin{itemize}
    \item Federated averaging assumes roughly similar data across clients
    \item Real deployments show heavy label and feature skew per client
    \item Naive FedAvg diverges or converges slowly under strong skew
  \end{itemize}
\end{frame}

\begin{frame}{Related Approaches}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}

      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Method}

\begin{frame}{Our Approach}
  \begin{itemize}
    \item Adaptive client weighting based on local gradient variance
    \item Periodic re-clustering as client distributions drift over time
    \item No extra communication round beyond standard FedAvg
  \end{itemize}
\end{frame}

\begin{frame}{Weighting Rule}
  Client $i$'s contribution weight at round $t$:
  \[
    w_i^{(t)} = \frac{1 / \sigma_i^{(t)}}{\sum_{j} 1 / \sigma_j^{(t)}}
  \]
  where $\sigma_i^{(t)}$ is the local gradient variance estimated from the
  last three local epochs.
\end{frame}

\section{Conclusion}

\begin{frame}{Summary}
  \begin{itemize}
    \item Variance-based weighting outperforms uniform and loss-based baselines
    \item Gains grow with skew severity
    \item No additional communication cost over standard FedAvg
