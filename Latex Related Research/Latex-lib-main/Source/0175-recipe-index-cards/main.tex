\ingredient{2 cups all-purpose flour}
\ingredient{1 tsp baking soda}
\ingredient{1 tsp salt}
\ingredient{1 cup unsalted butter, softened}
\ingredient{3/4 cup granulated sugar}
\ingredient{3/4 cup packed brown sugar}
\ingredient{2 large eggs}
\ingredient{2 tsp vanilla extract}
\ingredient{2 cups chocolate chips}
\cookingtime{25 min.}
\cooktemp{375}{F}
\begin{recipe}{Chocolate Chip Cookies}{24 cookies}
Whisk the flour, baking soda and salt together in a bowl and set aside.

Cream the butter with both sugars until light and fluffy, about 3 minutes.
Beat in the eggs one at a time, then the vanilla.

Stir the dry ingredients into the butter mixture just until combined, then
fold in the chocolate chips.

Drop rounded tablespoons of dough onto ungreased baking sheets, spaced
2 inches apart. Bake until the edges are golden but the centers still
look slightly underdone.

Cool on the sheet for 5 minutes before transferring to a wire rack.
\end{recipe}

\ingredient{1 lb spaghetti}
\ingredient{4 large eggs}
\ingredient{1 cup finely grated Pecorino Romano}
\ingredient{8 oz pancetta, diced}
\ingredient{black pepper, to taste}
\cookingtime{20 min.}
\begin{recipe}{Spaghetti Carbonara}{4}
Bring a large pot of salted water to a boil and cook the spaghetti until
al dente. Reserve 1 cup of the pasta water before draining.

While the pasta cooks, render the pancetta in a large skillet over medium
heat until crisp, about 6 minutes.

Whisk the eggs and cheese together in a bowl with a generous amount of
black pepper.

Off the heat, toss the hot, drained pasta with the pancetta, then quickly
stir in the egg mixture, adding splashes of the reserved pasta water until
glossy and creamy. Serve immediately with extra cheese and pepper.
\end{recipe}
