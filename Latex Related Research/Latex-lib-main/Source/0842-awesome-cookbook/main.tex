\category{Starters}
\recipe{Citrus and Herb Salad}
\info[servings=4,time=15,source=Kitchen notes]{}
\begin{ingredients}
  \ingredient{2}{}{oranges, peeled and sliced}
  \ingredient{1}{}{fennel bulb, thinly sliced}
  \ingredient{2}{tbsp}{olive oil}
  \ingredient{1}{tbsp}{lemon juice}
\end{ingredients}
\begin{preparation}
  \step Arrange the citrus and fennel on a serving plate.
  \step Whisk the oil and lemon juice, season to taste, and spoon over the salad.
  \step Finish with chopped herbs and serve immediately.
\end{preparation}
\category{Main Courses}
\recipe{Weeknight Tomato Pasta}
\info[servings=2,time=30]{}
\begin{ingredients}
  \ingredient{200}{g}{pasta}
  \ingredient{400}{g}{tomatoes}
  \ingredient{1}{}{garlic clove}
\end{ingredients}
\begin{preparation}
  \step Cook the pasta until just tender and reserve a cup of its water.
  \step Simmer the tomatoes and garlic while the pasta cooks.
  \step Toss together, adding pasta water until glossy, and serve.
\end{preparation}
