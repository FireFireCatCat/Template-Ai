\section{Why Drones}

\begin{frame}{The inspection backlog}
  \begin{itemize}
    \item 340 bridges are due a detailed inspection this year
    \item Manual rope-access inspection covers roughly 40 bridges a year
    \item Drone-assisted inspection is the only way to close the gap
  \end{itemize}
\end{frame}

\subsection{Pilot Design}

\begin{frame}{Twelve crossings, two methods}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Manual}
      \begin{itemize}
        \item Rope-access teams
        \item Full tactile inspection
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Drone}
      \begin{itemize}
        \item High-res visual + thermal
        \item No lane closures needed
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Findings}

\begin{frame}{Defects found, by method}
  \begin{table}
    \centering
    \begin{tabular}{lcc}
      \toprule
      Defect class & Manual & Drone \\
      \midrule
      Surface cracking   & 28 & 31 \\
      Corrosion staining & 19 & 22 \\
      Joint seal failure & 11 & 9  \\
      \bottomrule
    \end{tabular}
  \end{table}
  Drone inspection found more surface-level defects; manual inspection remained better at joint seals hidden from camera angles.
\end{frame}

\section{Recommendation}

\begin{frame}{Where we land}
  \begin{enumerate}
    \item Use drones for the first-pass screening on all 340 bridges
    \item Reserve rope-access teams for joints and confirmed defects
    \item Re-evaluate lidar-based crack-width measurement next cycle
  \end{enumerate}
\end{frame}
