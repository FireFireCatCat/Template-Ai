\begin{frame}
  \frametitle{Why a class, not a package}
  \begin{itemize}[<+->]
    \item Beamer builds presentations on top of the article class
    \item ltx-talk is written from scratch as a presentation class
    \item That lets it produce properly tagged, accessible PDF output
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{What you get}
  \begin{itemize}
    \item Structured headings and reading order
    \item Overlay support without hand-rolled tricks
    \item A footer that tracks title, author and frame number
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Try it yourself}
  Install from CTAN or TeX Live, then compile with pdflatex, LuaLaTeX or
  XeLaTeX --- the class does not require a specific engine.
\end{frame}
