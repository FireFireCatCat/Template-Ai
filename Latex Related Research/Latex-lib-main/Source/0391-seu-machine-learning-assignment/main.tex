\problem{Linearly Separable SVM}
\subproblem{}
Given a training set $\{(\mathbf{x}_i, y_i)\}_{i=1}^n$ with
$y_i \in \{-1,+1\}$, write the primal optimization problem for the
hard-margin support vector machine.

\begin{proof}
The hard-margin SVM primal problem is
\begin{equation}
    \min_{\mathbf{w},b} \ \frac{1}{2}\|\mathbf{w}\|^2
    \quad \text{s.t.} \quad y_i(\mathbf{w}^\top\mathbf{x}_i + b) \ge 1,\ \forall i,
\end{equation}
which correctly classifies every training point while maximizing the
geometric margin $2/\|\mathbf{w}\|$.
\end{proof}

\subproblem{}
Write the corresponding Lagrangian dual problem.

\begin{proof}
Introducing multipliers $\alpha_i \ge 0$, the dual problem is
\begin{equation}
    \max_{\bm{\alpha}} \ \sum_{i=1}^n \alpha_i
    - \frac{1}{2}\sum_{i=1}^n\sum_{j=1}^n \alpha_i\alpha_j y_i y_j \mathbf{x}_i^\top\mathbf{x}_j
    \quad \text{s.t.} \quad \sum_{i=1}^n \alpha_i y_i = 0,\ \alpha_i \ge 0.
\end{equation}
The optimum satisfies $\mathbf{w}^* = \sum_i \alpha_i^* y_i \mathbf{x}_i$,
with support vectors corresponding to $\alpha_i^* > 0$.
\end{proof}

\problem{Kernel Methods}
\subproblem{}
Prove that the Gaussian kernel
$K(\mathbf{x},\mathbf{z}) = \exp(-\|\mathbf{x}-\mathbf{z}\|^2/2\sigma^2)$
is a valid kernel (its Gram matrix is positive semi-definite).

\begin{proof}
Expand the Gaussian kernel as
\begin{equation}
    K(\mathbf{x},\mathbf{z}) = \exp\left(-\frac{\|\mathbf{x}\|^2}{2\sigma^2}\right)
    \exp\left(\frac{\mathbf{x}^\top\mathbf{z}}{\sigma^2}\right)
    \exp\left(-\frac{\|\mathbf{z}\|^2}{2\sigma^2}\right).
\end{equation}
The middle factor $\exp(\mathbf{x}^\top\mathbf{z}/\sigma^2)$ expands as
an infinite series of polynomial kernels, each term of which is
positive semi-definite; nonnegative combinations and pointwise
(Schur) products of PSD kernels remain PSD, and the outer positive
scalar factors preserve PSD-ness, so $K$ is a valid kernel.
\end{proof}
