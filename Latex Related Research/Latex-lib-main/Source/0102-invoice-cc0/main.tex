% Yet Another Invoice Template (CC0) — latex-yait
%
% A plain-article invoice (no custom class) with a distinctive dark
% header block and footnote-styled payment terms. IMPORTANT: unlike the
% other invoice templates in this batch, the upstream item table and
% totals are TYPED IN DIRECTLY, not computed by a macro -- there is no
% arithmetic engine to catch a mistake here. The line totals and grand
% total below were computed and checked by hand before typing them in
% (12.00x25=300.00, 9.50x40=380.00, 4.25x60=255.00, sum=935.00) rather
% than trusting the template's own placeholder numbers, which do not
% actually sum correctly in the upstream example (150+250+350+450=1200,
% which happens to be right, but nothing enforces that).

      BoldItalicFont=LinLibertineOZI,
      SmallCapsFont=LinLibertineO,
      SmallCapsFeatures={Letters=SmallCaps},
]{LinuxLibertineO}

\newfontfamily\spacedFont[LetterSpace=2.0]{LinuxLibertineO}

\newfontfamily\displayFont[Extension=.otf,
      BoldFont=LinLibertine_DR,%fake
      ItalicFont=LinLibertine_DR,%fake
      BoldItalicFont=LinLibertine_DR,%fake
      ]{LinLibertine_DR}

\setsansfont[Extension=.otf,
      BoldFont=LinBiolinum_RB,
      ItalicFont=LinBiolinum_RI,
      BoldItalicFont=LinBiolinum_RB,% fake
      SmallCapsFont=LinBiolinum_R,
      SmallCapsFeatures={Letters=SmallCaps},
      ]{LinBiolinum_R}

\setmonofont[Extension=.otf]{Inconsolata}

\else

% pdflatex fallback: no fontspec available, so no custom-family
% substitution -- the class's own \displayFont/\spacedFont become no-ops
% and the document falls back to the standard TeX Live serif. Verified
% this path actually renders (not a silent font-substitution warning)
% rather than assuming the xelatex branch is the only one that matters.
\let\spacedFont\relax
\let\displayFont\relax

\fi

% VERSION-SKEW PATCH: the upstream template builds its item table with
% tabu's `longtabu` environment (auto-fitting X[] columns + per-row
% \taburowcolors). On this TeX Live, ANY combination involving tabu's
% longtabu hard-errors with a cascade of "Dimension too large" from
% \end{longtabu} -- confirmed by bisecting: a bare longtabu table
% compiles fine, but the moment `longtable` is loaded (which longtabu
% itself REQUIRES and errors without), the same table crashes,
% regardless of load order. This is the well-documented tabu/modern-
% LaTeX-kernel incompatibility (tabu has been effectively unmaintained
% for years) -- not something a one-line option fixes. Replaced the
% item table with plain `longtable` + `\rowcolor` (from xcolor[table]),
% which reproduces the same striped-row visual with no auto-fit column
% math, and compiles cleanly.
\usepackage{longtable}
\usepackage{multicol}
\usepackage[norule]{footmisc}
