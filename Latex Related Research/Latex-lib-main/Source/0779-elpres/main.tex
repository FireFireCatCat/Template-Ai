    \normalsize\mdseries Moving 40{,}000 vehicles to the new data pipeline

\begin{psli}[Why Migrate]
  \begin{citemize}
    \item The old telemetry pipeline drops packets under load
    \item No replay capability for post-incident analysis
    \item New pipeline adds exactly-once delivery guarantees
  \end{citemize}
\end{psli}

\begin{psli}[Migration Plan]
  \begin{cenumerate}
    \item Dual-write from vehicles to both pipelines
    \item Validate parity on a 5\% sample for two weeks
    \item Cut over reads, then retire the old pipeline
  \end{cenumerate}
\end{psli}

\begin{psli}[Packet Loss, Before and After]

      Regional & 1.8\% & 0.02\% \\
    \end{tabular}
  \end{center}
\end{psli}
