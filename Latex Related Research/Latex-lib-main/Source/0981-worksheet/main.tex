\begin{exercise}%
[\score{10}\easy\learningtargets{1, 2}\occurrence{Chapter 4, Section 2}]
Solve the following quadratic equation by factoring:
\[
  x^2 - 5x + 6 = 0.
\]
\end{exercise}

\begin{exercise}%
[\score{15}\medium\learningtargets{2, 3}\occurrence{Chapter 4, Section 3}]
Solve the following quadratic equation using the quadratic formula, and
give your answer to two decimal places:
\[
  2x^2 + 3x - 7 = 0.
\]
\end{exercise}

\begin{exercise}%
[\score{20}\hard\learningtargets{3, 4}\occurrence{Chapter 4, Section 4}]
A ball is thrown upward from a height of $2\,\text{m}$ with an initial
velocity of $15\,\text{m/s}$. Its height in metres after $t$ seconds is
given by
\[
  h(t) = -4.9t^2 + 15t + 2.
\]
Determine, to the nearest tenth of a second, how long the ball takes to
hit the ground.
\end{exercise}

\begin{exercise}%
[\score{5}\easy\learningtargets{1}\occurrence{Chapter 4, Section 1}]
Without solving it, determine the number of real solutions of
\[
  x^2 + 4x + 7 = 0
\]
by evaluating its discriminant.
\end{exercise}
