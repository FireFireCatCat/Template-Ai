\documentclass[journal]{IEEEtran}

\usepackage{amsmath}
\usepackage{url}
\usepackage{graphicx}
\usepackage{float}

\begin{document}

\title{Term Paper Title}
\author{Student Name}
\maketitle

\begin{abstract}
A short summary of the paper's topic, method, and findings, written
after the rest of the paper is drafted.
\end{abstract}

\section{Introduction}
Describe the planned activities and the specific question the paper
investigates, with references to the sources that motivate it
\cite{shannon1948}.

\section{Background}
Give the topic background, citing prior work as it becomes
relevant~\cite{turing1950}.

\section{Results}
Present the findings from interviews, observations, or experiments.

\section{Discussion}
Evaluate and comment on the results presented above.

\section{Summary}
Summarise the paper's contribution and any open questions it leaves.

\appendix
\section{Supporting Material}
Include interview subject details, survey questions, or other
supporting material that would otherwise interrupt the main text.

\bibliographystyle{IEEEtran}
\bibliography{references}

\end{document}
