% vienna-cs-thesis — a University of Vienna Faculty of Computer
% Science-style thesis, on plain `report`. The upstream template's own
% custom UniVieCS_Thesis class (title-page variants, glossaries package
% wired to an external makeglossaries pass) is not vendored — see
% README.md — so this starter reproduces the front matter / chapter /
% back matter shape by hand, including a plain acronym table in place
% of the glossaries package (this pipeline runs no makeglossaries pass,
% so a real glossaries-package acronym list would render empty).
\documentclass[a4paper,11pt]{report}

\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
% english loaded LAST so it is babel's default/main language — babel
% activates the last-listed language outside explicit \selectlanguage
% blocks. Loading ngerman last leaked a German-formatted \today onto
% the (English) title page in an earlier draft of this starter.
\usepackage[ngerman,english]{babel}
\usepackage{microtype}
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{algorithm}
\usepackage{algpseudocode}
\usepackage{listings}
\usepackage[hidelinks]{hyperref}

\title{Thesis Title}
\author{Student Name}
\date{\today}
